### 阶段2: 创建第一个行为树工程
我们做一个小项目:
> 巡检机器人行为树:
> 正常情况下: 去A点巡检 ----> 去B点巡检----> 回Dock
> 如果电量低: 立即打断当前任务 -----> 返回Dock

创建包:
```
cd ~/ros2_bt_ws/src
ros2 pkg create bt_patrol_demo \
	--buuld-type ament_cmake \
	--dependencies rclcpp std_msgs behaviortree_cpp \
	--license Apache-2.0
```
进入包目录:
```
cd ~/ros2_bt_ws/src/bt_patrol_demo
mkdir -p bt_xml
```
### 阶段3: 写行为树XML
创建XML文件:
```
cat > bt_xml/patrol.xml <<'EOF'
<root BTCPP_format="4" main_tree_to_execute="MainTree">
	<BehaviorTree ID="MainTree">
		<ReactiveFallback name="mission_or_return_to_dock">
			<!-- 如果电量低, 立刻回Dock -->
			<Sequence name="low_battery_return">
				<BatteryLow battery="{battery_level}" threshold="0.25"/>
				<MoveTo target="Dock"/>
			</Sequence>
			
			<!--正常巡检任务-->
			<Sequence name="normal_patrol">
				<MoveTo target="A"/>
				<Inspect target="A" seconds="10"/>
				<MoveTo target="B"/>
				<Inspect target="B" seconds="10"/>
				<MoveTo target="Dock"/>
			</Sequence>
		<ReactiveFallback>
	</BehaviorTree>
</root>
EOF
```
这里最重要的是:
```
<BatteryLow battery="{battery_level}" threshold="0.25"/>
```
{battery_level} 是黑板变量. BehaviorTree.CPP里, Blackboard是所有节点共享的键值存储, Input Port可以读取黑板里的值.

### 阶段4: 写C++行为树节点
创建源码文件:
```
cat > src/ros_patrol_bt_node.cpp <<'EOF'
#include <algorithm>
#include <chrono>
#include <iostream>
#include <string>
#include "rclcpp/rclcpp.hpp"
#include "std_msgs/msg/float32.hpp"
#include "behaviortree_cpp/bt_factory.h"

using namespace std::chrono_literals;
static std::string statusToString(BT::NodeStatus status)
{
	switch(status)
	{
		case BT::NodeStatus::SUCCESS:
			return "SUCCESS";
		case BT::NodeStatus::FAILURE:
			return "FAILURE";
		case BT::NodeStatus::RUNNING:
			return "RUNNING";
	}
	return "UNKNOWN";
}

// 条件节点: 电量是否低
class BatteryLow : public BT::ConditionNode
{
public:
	BatteryLow(const std::string& name, const BT::NodeConfig& config)
	: BT::ConditionNode(name,config)
	{
	}
	
	static BT::PortsList providedPorts()
	{
		return {
			BT::InputPort<double>("battery"),
			BT::InputPort<double>("thresold")
		};
	}

	BT::NodeStatus tick() override
	{
		double battery = 1.0;
		double thresold = 0.25;
		if(!getInput<double>("battery",battery))
		{
			throw BT::RuntimeError("BatteryLow missing input [battery]");
		}
		if(!getInput<double>("thresold",thresold))
		{
			throw BT::RuntimeError("BatteryLow missing input [thresold]");
		}
		const bool low = battery < thresold;
		std::cout <<"[BatteryLow] battery="<<battery
				  << ", thresold="<<thresold 
				  <<"->" <<(low?"SUCCESS":"FAILURE")
				  <<std::endl;
		return low?BT::NodeStatus::SUCCESS : BT::NodeStatus::FAILURE;
	}

};

// 同步动作节点: 模拟移动到某个目标点
class MoveTo : public BT::SyncActionNode
{
public: 
	MoveTo(const std::string& name, const BT::NodeConfig& config)
	:  BT::SyncActionNode(name, config)
	{
	}
	static BT::PortsList providedPorts()
	{
		return {
			BT::InputPort<std::string>("target")
		};
	
	}
	BT::NodeStatus tick() override 
	{
		std::string target;
		if(!getInput<std::string>("target",target))
		{
			throw BT::RuntimeError("MoveTo missing input [target]");
		}	
		std::cout<<"[MoveTo] moving to "<<target<<std::endl;
		// 这里先模拟移动, 后面这个节点会被替换成 ROS2 Action/ Nav2 NavigateToPose
		std::this_thread::sleep_for(500ms);
		std::cout<<"[MoveTo] arrived at "<<target<<std::endl;
		return BT::NodeStatus::SUCCESS;
	}
};
// 有状态动作节点: 模拟巡检,需要持续RUNNING 一段时间
class Inspect : public BT::StatefulActionNode
{
public:
	Inspect(const std::string& name, const BT::NodeConfig& config)
	: BT::StatefulActionNode(name,config){
	}
	static BT::PortsList providedPorts()
	{
		return {
			BT::InputPort<std::string>("target"),
			BT::InputPort<int>("seconds")
		};
	}
	BT::NodeStatus onStart() override
	{
		int seconds =3;
		if(!getInput<std::string>("target",target_))
		{
			throw BT::RuntimeError("Inspect missing input [target]");
		}
		if(!getInput<int>("seconds",seconds))
		{
			throw BT::RuntimeError("Inspect missing input [seconds]");
		}
		start_time_ = std::chrono::steady_clock::now();
		end_time_ = start_time_ +std::chrono::seconds(seconds);
		last_print_tiem_ = start_time_;
		std::cout<<"[Inspect] start inspecting "<<target_ 
				<< " for "<<seconds<<" seconds" <<std::endl;
		return BT::NodeStatus::RUNNING;
	}
	
	BT::NodeStatus onRunning() override
	{
		const auto now = std::chrono::steady_clock::now();
		if(now>=end_time_){
			std::cout<<"[Inspect] finished inspecting "<<target_<<std::endl;
			return BT::NodeStatus::SUCCESS;
		}
		if(now-last_print_time_ >= 1s)
		{
			std::cout<<"[Inspect] working at " <<target_<<std::endl;
			return BT::NodeStatus::RUNNING;
		
		}

		return BT::NodeStatus::RUNNING;	
	}
	void onHalted() override 
	{
		std::cout<<"[Inpsect] halted at "<<target_
				 <<"because behavior tree switched branch"
				 << std::endl;
	}
private: 
	std::string target_;
	std::chrono::steady_clock::time_point start_time_;
	std::chrono::steady_clock::time_point end_time_;
	std::chrono::steady_clock::time_point last_print_time_;

};
int main(int argc, char** argv)
{
	rclcpp::int(argc,argv);
	if(argc < 2){
		std::cerr << "Usage:\n"<<
		"ros2 run bt_patrol_demo ros_patrol_bt_node <path_to_bt_xml>\n";
		rclcpp::shutdown();
		return 1; 
	}
	const std::string xml_file = argv[1];
	auto node = rclcpp::Node::make_shared("ros_patrol_bt_node");
	//Blackboard: 行为树共享数据区
	auto blackboard = BT::Blackboard::create();
	blackboard->set("battery_level",1.0);
	auto battery_sub = node->create_subscription<std_msgs::msg::Float32>(
		"/battery_level",
		10,
		[blackboard,node](const std_msgs::msg::Float32::SharedPtr msg){
			const double raw = static_cast<double>(msg->data);
			const double battery = std::clamp(raw,0.0,1.0);
			blackboard->set("battery_level",battery);
			RCLCPP_INFO(
				node->get_logger(),
				"battery_level update: %.2f",
				battery
			);
		}
	
	);
	BT::BehaviorTreeFactory factory;
	
	factory.registerNodeType<BatteryLow>("BatteryLow");
	factory.registerNodeType<MoveTo>("MoveTo");
	factory.registerNodeType<Inspect>("Inspect");
	
	auto tree = factory.createTreeFromFile(xml_file,blackboard);
	rclcpp::executors::SingleThreadedExecutor executor;
	executor.add_node(node);
	
	BT::NodeStatus status = BT::NodeStatus::RUNNING;
	RCLCPP_INFO(node->get_logger(),"Behavior Tree Started.");
	
	while(rclcpp::ok() && status == BT::NodeStatus::RUNNING)
	{
		executor.spin_some();
		status = tree.tickOnce();
		std::cout<<"[Tree] status = "<< statusToString(status)<<std::endl;
		if(status == BT::NodeStatus::RUNNING)
		{
			tree.sleep(100ms);
		
		}
	}
	RCLCPP_INFO(
		node->get_logger(),
		"Behavior Tree finished with status:%s",
		statusToString(status).c_str()
	);
	rclcpp::shutdonw();
	return status == BT::NodeStatus::SUCCESS ? 0 : 2;
	
}
EOF 
```
这里明确三种节点
```
BatteryLow -> ConditionNode 只判断 不干活
MoveTo     -> SyncActionNode 同步动作,执行完才返回
Inspect    -> StatefulActionNode, 能返回RUNNING,可被打断
```
行为树里的节点tick后必须返回SUCCESS、FAILURE、RUNNING; RUNNING表示动作还没完成, 需要后续继续tick. BehaviorTree.CPP官方也推荐长任务用StatefulActionNode, 因为它可以在onStart、onRunning、onHalted 中表达异步/可中断行为.
