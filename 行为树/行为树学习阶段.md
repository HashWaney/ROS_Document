### 阶段0 : 环境确认
打开终端, 先执行
```
source /opt/ros/jazzy/setup.bash
echo $ROS_DISTRO
ros2 pkg list | sed -n '1,10p'
```
应该可以看到
```
jazzy
```
然后安装本阶段需要的包:
```
sudo apt update
sudo apt install -y \
	python3-colcon-common-extensions \
	ros-jazzy-behaviortree-cpp
```
确认BehaviorTree.CPP包存在:
```
ros2 pkg list | grep behaviortree
```

### 阶段2: 创建第一个行为树工程
我们做一个小项目:
> 巡检机器人行为树:
> 正常情况下: 去A点巡检 ----> 去B点巡检----> 回Dock
> 如果电量低: 立即打断当前任务 -----> 返回Dock

创建包:
```
cd ~/ros2_bt_ws/src
ros2 pkg create bt_patrol_demo \
	--buuld-type ament_cmake \
	--dependencies rclcpp std_msgs behaviortree_cpp \
	--license Apache-2.0
```
进入包目录:
```
cd ~/ros2_bt_ws/src/bt_patrol_demo
mkdir -p bt_xml
```
### 阶段3: 写行为树XML
创建XML文件:
```
cat > bt_xml/patrol.xml <<'EOF'
<root BTCPP_format="4" main_tree_to_execute="MainTree">
	<BehaviorTree ID="MainTree">
		<ReactiveFallback name="mission_or_return_to_dock">
			<!-- 如果电量低, 立刻回Dock -->
			<Sequence name="low_battery_return">
				<BatteryLow battery="{battery_level}" threshold="0.25"/>
				<MoveTo target="Dock"/>
			</Sequence>
			
			<!--正常巡检任务-->
			<Sequence name="normal_patrol">
				<MoveTo target="A"/>
				<Inspect target="A" seconds="10"/>
				<MoveTo target="B"/>
				<Inspect target="B" seconds="10"/>
				<MoveTo target="Dock"/>
			</Sequence>
		<ReactiveFallback>
	</BehaviorTree>
</root>
EOF
```
这里最重要的是:
```
<BatteryLow battery="{battery_level}" threshold="0.25"/>
```
{battery_level} 是黑板变量. BehaviorTree.CPP里, Blackboard是所有节点共享的键值存储, Input Port可以读取黑板里的值.

### 阶段4: 写C++行为树节点
创建源码文件:
```
cat > src/ros_patrol_bt_node.cpp <<'EOF'
#include <algorithm>
#include <chrono>
#include <iostream>
#include <string>
#include "rclcpp/rclcpp.hpp"
#include "std_msgs/msg/float32.hpp"
#include "behaviortree_cpp/bt_factory.h"

using namespace std::chrono_literals;
static std::string statusToString(BT::NodeStatus status)
{
	switch(status)
	{
		case BT::NodeStatus::SUCCESS:
			return "SUCCESS";
		case BT::NodeStatus::FAILURE:
			return "FAILURE";
		case BT::NodeStatus::RUNNING:
			return "RUNNING";
	}
	return "UNKNOWN";
}

// 条件节点: 电量是否低
class BatteryLow : public BT::ConditionNode
{
public:
	BatteryLow(const std::string& name, const BT::NodeConfig& config)
	: BT::ConditionNode(name,config)
	{
	}
	
	static BT::PortsList providedPorts()
	{
		return {
			BT::InputPort<double>("battery"),
			BT::InputPort<double>("thresold")
		};
	}

	BT::NodeStatus tick() override
	{
		double battery = 1.0;
		double thresold = 0.25;
		if(!getInput<double>("battery",battery))
		{
			throw BT::RuntimeError("BatteryLow missing input [battery]");
		}
		if(!getInput<double>("thresold",thresold))
		{
			throw BT::RuntimeError("BatteryLow missing input [thresold]");
		}
		const bool low = battery < thresold;
		std::cout <<"[BatteryLow] battery="<<battery
				  << ", thresold="<<thresold 
				  <<"->" <<(low?"SUCCESS":"FAILURE")
				  <<std::endl;
		return low?BT::NodeStatus::SUCCESS : BT::NodeStatus::FAILURE;
	}

};

// 同步动作节点: 模拟移动到某个目标点
class MoveTo : public BT::SyncActionNode
{
public: 
	MoveTo(const std::string& name, const BT::NodeConfig& config)
	:  BT::SyncActionNode(name, config)
	{
	}
	static BT::PortsList providedPorts()
	{
		return {
			BT::InputPort<std::string>("target")
		};
	
	}
	BT::NodeStatus tick() override 
	{
		std::string target;
		if(!getInput<std::string>("target",target))
		{
			throw BT::RuntimeError("MoveTo missing input [target]");
		}	
		
	}

}
```
